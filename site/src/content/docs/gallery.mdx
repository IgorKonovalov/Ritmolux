import Frame from '../../components/Frame.astro';

Every preset that ships, one frame each, grouped by the system it draws with. Each frame is the
same capture: the synthesized `dynamic:110` clip at its loudest hop, at the Rich quality tier —
the settings [`scripts/docs-shots.mjs`](../../../../scripts/docs-shots.mjs)
records for every committed picture in this repository.

**Click a frame to open it full size without leaving this page; click the name under it — or
the link inside that window — to read the preset that drew it.** A preset is a text file —
[the preset guide](../../../../docs/preset-guide.md) opens the smallest one worth looking at,
and [the parameter roster](../../../../presets/README.md) lists every named parameter each of
these systems takes.

{/*
  Where a frame's CAPTION links to - the picture itself opens at full size in a
  modal, which `Frame` handles, and that modal offers this same destination
  under `detailLabel`. A preset has no page of its own on this site: it IS a
  text file, so the file is the destination. `blob/main` rather than a tag,
  because the gallery is rebuilt from the same tree it links into.
*/}

export const SOURCE = 'https://github.com/IgorKonovalov/Ritmolux/blob/main/presets/';

export const cards = Object.entries(
  import.meta.glob('../../../../docs/images/gallery/presets/*.png', {
    eager: true,
    import: 'default',
  }),
).map(([file, image]) => ({ image, preset: file.split('/').pop().replace('.png', '') }));

export const systemOf = Object.fromEntries(
  Object.entries(
    import.meta.glob('../../../../presets/*.toml', {
      eager: true,
      query: '?raw',
      import: 'default',
    }),
  ).map(([file, text]) => [
    file.split('/').pop().replace('.toml', ''),
    /^system\s*=\s*"(.+?)"/m.exec(text)?.[1] ?? 'unknown',
  ]),
);

export const families = [
  ...cards.reduce((acc, card) => {
    const system = systemOf[card.preset] ?? 'unknown';
    acc.set(system, [...(acc.get(system) ?? []), card]);
    return acc;
  }, new Map()),
]
  .map(([system, members]) => [system, members.sort((a, b) => a.preset.localeCompare(b.preset))])
  .sort(([a], [b]) => a.localeCompare(b));

<ul class="family-index">
  {families.map(([system, members]) => (
    <li><a href={`#${system}`}><code>{system}</code> {members.length}</a></li>
  ))}
</ul>

{families.map(([system, members]) => (
  <section class="family-section">
    <h2 id={system}>
      <code>{system}</code> <span class="family-count">{members.length} presets</span>
    </h2>
    <div class="gallery-strip" role="list">
      {members.map(({ image, preset }) => (
        <Frame
          image={image}
          href={`${SOURCE}${preset}.toml`}
          label={preset}
          detailLabel="Read the preset"
          alt={`The ${preset} preset, drawn with the ${system} system`}
          sizes="(min-width: 46rem) 13rem, 45vw"
        />
      ))}
    </div>
  </section>
))}
