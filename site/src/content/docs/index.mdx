  subpath out - and this site is built at two different ones (`/ritmolux/` for the
  demo, `/Ritmolux/` for the project deployment). `import.meta.env.BASE_URL` is
  whichever one this build used, and it always ends in a slash.

export const gallery = Object.entries(
  import.meta.glob('../../../../docs/images/gallery/*.png', { eager: true, import: 'default' }),
)
  .map(([file, image]) => ({ image, system: file.split('/').pop().replace('.png', '') }))
  .sort((a, b) => a.system.localeCompare(b.system));
